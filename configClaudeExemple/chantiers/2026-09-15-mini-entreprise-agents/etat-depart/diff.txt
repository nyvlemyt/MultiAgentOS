+class IssRatingLastModificationUploadTests(ApiE2ETestCase):
+    """
+    `iss_esg_rating_last_modification` is declared `date` by pandera and stored in a
+    `DateField`: whatever format the source writes the value in, what the database gives
+    back is a `date`, parsed once at integration time.
+
+    It used to be a `CharField` fed by the string branch of the cast, which stored the
+    source text verbatim and left every reader to parse it again. The freshness sheet of
+    the quality report already parsed it on the fly.
+
+    These tests read the stored value because that is where the change acts. The export
+    tests in `test_export_demain.py` cover the contract with DEMAIN, one layer further
+    down: a value already in ISO would come out of the export unchanged even if the field
+    went back to a `CharField`.
+    """
+
+    def stored_rating_date(self, isin: str) -> date | None:
+        """The rating date as the database gives it back, for the issuer holding `isin`."""
+        return ISSEquityEsgIssuerData.objects.get(isin=isin).iss_esg_rating_last_modification
+
+    def test_a_source_date_written_in_another_format_is_stored_as_a_date(self) -> None:
+        """The normalisation case: a European form is parsed, not kept as text."""
+        response = self.post_insert("iss_equity_esg_issuer_data", [
+            {"ISIN": "FR0000000001", "ISSESGRatingLastmodification": "31/12/2024"},
+        ])
+
+        self.assertEqual(200, response.status_code)
+        self.assertEqual(date(2024, 12, 31), self.stored_rating_date("FR0000000001"))
+
+    def test_a_source_date_already_in_iso_is_stored_as_a_date(self) -> None:
+        """
+        The typing case: the value comes back as a `date`, not as the ISO text it was
+        written in. This is what a return to a `CharField` would break, and what the
+        export test on the same value cannot see.
+        """
+        response = self.post_insert("iss_equity_esg_issuer_data", [
+            {"ISIN": "FR0000000001", "ISSESGRatingLastmodification": "2024-12-31"},
+        ])
+
+        self.assertEqual(200, response.status_code)
+        self.assertEqual(date(2024, 12, 31), self.stored_rating_date("FR0000000001"))
+
+    def test_a_sentinel_value_is_stored_as_null(self) -> None:
+        """
+        ISS writes `Not Collected` and `Not Applicable` where it has no date. The
+        preprocessing turns them into nulls before any cast, so the date branch never
+        sees them and the row is kept.
+        """
+        response = self.post_insert("iss_equity_esg_issuer_data", [
+            {"ISIN": "FR0000000001", "ISSESGRatingLastmodification": "Not Collected"},
+            {"ISIN": "FR0000000002", "ISSESGRatingLastmodification": "Not Applicable"},
+            {"ISIN": "FR0000000003", "ISSESGRatingLastmodification": "2024-12-31"},
+        ])
+
+        self.assertEqual(200, response.status_code)
+        self.assertIsNone(self.stored_rating_date("FR0000000001"))
+        self.assertIsNone(self.stored_rating_date("FR0000000002"))
+        # The third issuer is the control: without it, a column left entirely null for
+        # any other reason would satisfy the two assertions above. It also covers the
+        # production case, where one upload carries both sentinels and real dates.
+        self.assertEqual(date(2024, 12, 31), self.stored_rating_date("FR0000000003"))
+
+    def test_a_date_in_an_unsupported_format_is_stored_as_null_known_defect(self) -> None:
+        """
+        known_defect: a format none of the sixteen patterns of `parse_date_expr` covers is
+        stored as NULL, silently. The row survives, the date is lost, and nothing in the
+        response says so.
+
+        This is a loss introduced by the typing: the string branch used to store the text
+        verbatim and hand it over to the DEMAIN export as it was. No value in EveDev was
+        concerned on 07/09/2026 (0 out of 12 366 non-null values). The test pins the
+        behaviour so a change of source format is noticed here rather than downstream.
+        """
+        response = self.post_insert("iss_equity_esg_issuer_data", [
+            {"ISIN": "FR0000000001", "ISSESGRatingLastmodification": "31.12.2024"},
+        ])
+
+        self.assertEqual(200, response.status_code)
+        self.assertEqual(1, ISSEquityEsgIssuerData.objects.count())
+        self.assertIsNone(self.stored_rating_date("FR0000000001"))
